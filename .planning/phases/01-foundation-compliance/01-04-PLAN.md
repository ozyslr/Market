    rules_version = '2';
    service cloud.firestore {
      match /databases/{database}/documents {

        // --- HELPERS ---------------------------------------------------------
        function isSignedIn() { return request.auth != null; }
        function isAdmin() { return isSignedIn() && request.auth.token.role == 'admin'; }
        function isSeller() { return isSignedIn() && request.auth.token.role == 'seller'; }
        function isBuyer() { return isSignedIn() && request.auth.token.role == 'buyer'; }
        function getUserId() { return request.auth.uid; }

        // --- DEFAULT DENY ----------------------------------------------------
        match /{document=**} { allow read, write: if isAdmin(); }

        // --- SYSTEM ----------------------------------------------------------
        match /system/connection-test { allow read: if true; }

        // --- CATEGORIES ------------------------------------------------------
        match /categories/{cid} {
          allow read: if true;
          allow write: if isAdmin();
        }

        // --- USERS -----------------------------------------------------------
        match /users/{uid} {
          allow get: if isSignedIn() && (getUserId() == uid || isAdmin());
          allow list: if isAdmin();
          allow create: if isSignedIn() && uid == getUserId();
          allow update: if isSignedIn() && (
            (getUserId() == uid && request.resource.data.role == resource.data.role) || isAdmin()
          );
          allow delete: if false;    // use data deletion flow
        }

        // --- SELLERS ---------------------------------------------------------
        match /sellers/{sellerId} {
          allow read: if true;
          allow create: if isSignedIn() && (isAdmin() || request.resource.data.userId == getUserId());
          allow update: if isSignedIn() && (
            (resource.data.userId == getUserId() && request.resource.data.userId == resource.data.userId) || isAdmin()
          );
          allow delete: if isAdmin();
        }

        // --- PRODUCTS --------------------------------------------------------
        match /products/{pid} {
          allow read: if true;
          allow create: if isSignedIn() && (isAdmin() || isSeller());
          allow update: if isSignedIn() && (isAdmin() || (isSeller() && resource.data.sellerId == request.auth.token.sellerId));
          allow delete: if isAdmin() || (isSeller() && resource.data.sellerId == request.auth.token.sellerId);
        }

        // --- ORDERSETS (Plan 1) ----------------------------------------------
        match /orderSets/{orderSetId} {
          allow read: if isSignedIn() && (resource.data.userId == getUserId() || isAdmin());
          allow create: if isSignedIn() && request.resource.data.userId == getUserId();
          allow update: if isAdmin();
          allow delete: if false;
        }

        // --- SUBORDERS (Plan 1) ----------------------------------------------
        match /subOrders/{subOrderId} {
          allow read: if isAdmin();
          allow create: if false;
          allow update: if isAdmin();
          allow delete: if false;
        }

        // --- COMMISSION RULES (Plan 2) ---------------------------------------
        match /commissionRules/{ruleId} {
          allow read: if isSignedIn();
          allow write: if isAdmin();
        }

        // --- LEDGER (Plan 2 - Immutable) -------------------------------------
        match /ledger/{entryId} {
          allow read, create, update, delete: if false;   // server SDK only
        }

        // --- DATA DELETION REQUESTS (Plan 3) ---------------------------------
        match /dataDeletionRequests/{requestId} {
          allow read: if isAdmin();
          allow create: if isSignedIn();
          allow update: if isAdmin();
          allow delete: if false;
        }

        // --- CARTS -----------------------------------------------------------
        match /carts/{userId} {
          allow read, create, update, delete: if isSignedIn() && userId == getUserId();
        }

        // --- CART REMINDERS --------------------------------------------------
        match /cart_reminders/{reminderId} { allow read, write: if false; }

        // --- MAIL (Firebase Trigger Email) -----------------------------------
        match /mail/{mailId} {
          allow read, write: if false;
          allow create: if isSignedIn();
        }

        // --- EVENTS (analytics) ----------------------------------------------
        match /events/{eventId} {
          allow create: if isSignedIn();
          allow read: if isAdmin();
        }

        // --- CHAT SESSIONS ---------------------------------------------------
        match /chat_sessions/{sessionId} {
          allow read: if isSignedIn() && (resource.data.userId == getUserId() || isAdmin());
          allow create: if isSignedIn() && request.resource.data.userId == getUserId();
          allow update: if isSignedIn() && (resource.data.userId == getUserId() || isAdmin());
          allow delete: if isAdmin();
          match /messages/{messageId} {
            allow read: if isSignedIn();
            allow create: if isSignedIn();
            allow update: if isAdmin();
            allow delete: if false;
          }
        }

        // --- ACTIVITY LOGS ---------------------------------------------------
        match /activity_logs/{logId} {
          allow create: if isSignedIn();
          allow read: if isAdmin();

        match /orders/{oid} {
          allow read: if isSignedIn() && (resource.data.userId == getUserId() || resource.data.buyerId == getUserId() || isAdmin());
          allow create: if isSignedIn() && (request.resource.data.userId == getUserId() || request.resource.data.buyerId == getUserId());
          allow update: if isAdmin();
          allow delete: if false;
        }
      }
    }

<threat_model>

## Trust Boundaries

| Boundary               | Description                                                        |
| ---------------------- | ------------------------------------------------------------------ |
| client -> Firestore    | Direct Firestore reads/writes must respect per-collection rules    |
| client -> Express API  | Authenticated calls checked by verifyAdmin/verifySeller middleware |
| admin -> custom claims | setCustomUserClaims sets role claims that control all access       |

## STRIDE Threat Register

| Threat ID | Category               | Component                    | Disposition | Mitigation Plan                                                                                                   |
| --------- | ---------------------- | ---------------------------- | ----------- | ----------------------------------------------------------------------------------------------------------------- |
| T-01-016  | Elevation of Privilege | custom claims assignment     | mitigate    | Only verifyAdmin callers can set claims (admin-only endpoint)                                                     |
| T-01-017  | Elevation of Privilege | direct Firestore role bypass | mitigate    | Custom claims (request.auth.token.role) are set server-side via Admin SDK; client cannot modify their own claims  |
| T-01-018  | Information Disclosure | Firestore get() in rules     | mitigate    | Eliminated all get() calls -- use custom claims for role checks (also saves $ in billed reads)                    |
| T-01-019  | Tampering              | firestore.rules deployment   | accept      | Deployed via CLI with firebase deploy; no CI/CD enforcement in Phase 1                                            |
| T-01-SC   | Tampering              | npm packages                 | mitigate    | slopcheck + human verify checkpoint for any new packages; authMiddleware uses only existing deps (firebase-admin) |

</threat_model>
